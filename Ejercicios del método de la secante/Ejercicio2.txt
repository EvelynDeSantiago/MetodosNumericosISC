using System;

namespace Ejercicio2
{
    class Program
    {
        // Método de la secante para encontrar raíces
        public static double FindRoot(Func<double, double> function,
                                      double x0,
                                      double x1,
                                      double tolerance,
                                      int maxIterations)
        {
            double x2 = 0;
            int iterations = 0;

            do
            {
                x2 = x1 - ((function(x1) * (x1 - x0)) /
                      (function(x1) - function(x0)));

                x0 = x1;
                x1 = x2;

                iterations++;

            } while (Math.Abs(function(x2)) > tolerance &&
                     iterations < maxIterations);

            return x2;
        }

        static void Main(string[] args)
        {
            // Definir la función f(x) = x³ - 2x - 5
            Func<double, double> function = x => Math.Pow(x, 3) - 2 * x - 5;

            // Valores iniciales
            double x0 = 0.5;
            double x1 = 1.0;

            // Tolerancia y máximo de iteraciones
            double tolerance = 0.0001;
            int maxIterations = 1000;

            // Encontrar la raíz
            double root = FindRoot(function, x0, x1, tolerance, maxIterations);

            // Mostrar resultado
            Console.WriteLine("Raíz encontrada: " + root);

            // Mantener la consola abierta
            Console.ReadKey();
        }
    }
}
