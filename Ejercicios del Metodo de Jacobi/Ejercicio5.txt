using System;

class Ejercicio5
{
    static void Main(string[] args)
    {
        double[][] A =
        {
            new double[] {6, 1, -1},
            new double[] {1, 5, 1},
            new double[] {2, -1, 8}
        };

        double[] b = {12, 10, 20};

        double[] solucion = Jacobi(A, b);

        Console.WriteLine("------Resultado del sistema de ecuaciones:");
        Console.WriteLine("La solución aproximada del sistema de ecuaciones es:");

        for (int i = 0; i < solucion.Length; i++)
        {
            Console.WriteLine("x[" + i + "] = " + solucion[i].ToString("F4"));
        }
    }

    static double[] Jacobi(double[][] A, double[] b)
    {
        int n = A.Length;

        double[] x0 = new double[n];
        double[] xNuevo = new double[n];

        for (int k = 0; k < 50; k++)
        {
            for (int i = 0; i < n; i++)
            {
                double suma = 0;

                for (int j = 0; j < n; j++)
                {
                    if (j != i)
                    {
                        suma += A[i][j] * x0[j];
                    }
                }

                xNuevo[i] = (b[i] - suma) / A[i][i];
            }

            for (int i = 0; i < n; i++)
            {
                x0[i] = xNuevo[i];
            }
        }

        return x0;
    }
}
