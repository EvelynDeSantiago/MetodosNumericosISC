using System;

class Ejercicio4
{
    static void Main(string[] args)
    {
        double[][] matrix =
        {
            new double[] {1, 2, 3, 14},
            new double[] {2, -1, 1, 3},
            new double[] {3, 1, -1, 2}
        };

        GaussJordan(matrix);

        Console.WriteLine("---------Resultado del sistema de ecuaciones:");
        Console.WriteLine("La matriz en su forma escalonada reducida por filas es:");

        Imprimir(matrix);
    }

    static void GaussJordan(double[][] matrix)
    {
        int rows = matrix.Length;
        int cols = matrix[0].Length;

        for (int k = 0; k < rows; k++)
        {
            double divisor = matrix[k][k];

            for (int j = 0; j < cols; j++)
            {
                matrix[k][j] /= divisor;
            }

            for (int i = 0; i < rows; i++)
            {
                if (i != k)
                {
                    double factor = matrix[i][k];

                    for (int j = 0; j < cols; j++)
                    {
                        matrix[i][j] -= factor * matrix[k][j];
                    }
                }
            }
        }
    }

    static void Imprimir(double[][] matrix)
    {
        for (int i = 0; i < matrix.Length; i++)
        {
            for (int j = 0; j < matrix[0].Length; j++)
            {
                Console.Write(matrix[i][j] + "\t");
            }
            Console.WriteLine();
        }
    }
}

0	1	0	2
0	0	1	3
