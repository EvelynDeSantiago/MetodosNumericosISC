using System;

class Ejercicio3
{
    static void Main(string[] args)
    {
        double[][] A =
        {
            new double[] {10, 2, 1},
            new double[] {1, 5, 1},
            new double[] {2, 3, 10}
        };

        double[] b = {7, -8, 6};

        double[] x = GaussSeidel(A, b);

        Console.WriteLine("--------Resultado del sistema de ecuaciones:");
        Console.WriteLine("El resultado es:");

        for (int i = 0; i < x.Length; i++)
        {
            Console.WriteLine("x[" + i + "] = " + x[i].ToString("F4"));
        }
    }

    static double[] GaussSeidel(double[][] A, double[] b)
    {
        int n = b.Length;

        double[] x = new double[n];
        double[] xPrev = new double[n];

        while (true)
        {
            Array.Copy(x, xPrev, n);

            for (int i = 0; i < n; i++)
            {
                double suma = 0;

                for (int j = 0; j < n; j++)
                {
                    if (j != i)
                    {
                        suma += A[i][j] * x[j];
                    }
                }

                x[i] = (b[i] - suma) / A[i][i];
            }

            double maxDiff = 0;

            for (int i = 0; i < n; i++)
            {
                double diff = Math.Abs(x[i] - xPrev[i]);

                if (diff > maxDiff)
                {
                    maxDiff = diff;
                }
            }

            if (maxDiff < 0.0001)
            {
                break;
            }
        }

        return x;
    }
}
